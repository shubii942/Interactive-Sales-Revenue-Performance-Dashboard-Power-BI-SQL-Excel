import pandas as pd
from sqlalchemy import create_engine

# 1. Excel file name
excel_file = 'Interactive_Sales_Revenue_Performance_Data.xlsx'

# 2. Enter your MySQL Password here
db_user = 'root'
db_password = '99999'  # <-- Change this to your actual MySQL password
db_host = 'localhost'
db_port = '3306'
db_name = 'sales_performance'

# Create SQL Engine connection
engine = create_engine(f'mysql+pymysql://{db_user}:{db_password}@{db_host}:{db_port}/{db_name}')

# 3. Sheets to import
sheets = ['Fact_Sales', 'Dim_Products', 'Dim_Customers', 'Dim_Regions']

# 4. Upload sheets to MySQL
for sheet in sheets:
    df = pd.read_excel(excel_file, sheet_name=sheet)
    table_name = sheet.lower()
    df.to_sql(table_name, con=engine, if_exists='replace', index=False)
    print(f"✅ Sheet '{sheet}' successfully imported into table '{table_name}'")

print("\n🚀 All tables imported successfully into MySQL!")
